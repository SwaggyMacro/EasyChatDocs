import { HomeIcon } from 'lucide-react';
import { MonitorDot } from 'lucide-react';
import { TextCursorInput } from 'lucide-react';
import { SquarePen } from 'lucide-react';
import { Speech } from 'lucide-react';
import { TypeOutline } from 'lucide-react';
import { Languages } from 'lucide-react';
import { SpellCheck2 } from 'lucide-react';
import { MicSignal } from 'lucide-react';



<Cards>
  <Card
  icon={<MonitorDot />}
    href="./feature/screenshot-translation"
    title="截图翻译"
  >
    支持用户通过截图框选的方式自动识别并翻译截图中的文字，支持多种语言的翻译。
  </Card>

  <Card 
    icon={<SquarePen /> } 
    title="输入翻译" 
    href="./input-translation">
    用户输入中文或其他语言后，EasyChat 将自动识别并翻译为目标语言，并将翻译后的文本投递到目标应用中。
  </Card>

  <Card 
    icon={<TextCursorInput />} 
    href="./selection" 
    title="划词工具栏">
    支持划词翻译、总结、润色、纠正、解释等功能，用户可以在 `划词` 后选择所需的功能进行操作。
  </Card>

  <Card 
    icon={<Speech />} 
    href="./asr" 
    title="实时语音识别">
    支持实时语音识别功能，用户可通过 EasyChat 识别系统正在播放的 `音频/视频` 内容，并将识别结果翻译为目标语言作为字幕形式显示在屏幕上，支持 10 种语言的识别。
  </Card>

  <Card 
    icon={<TypeOutline />} 
    href="./image-ocr" 
    title="截图取词">
    支持用户通过截图框选的方式自动识别图片中的文字，并支持选择任意区域的文字进行操作，支持翻译、复制、解释以及图片翻译等功能。
  </Card>

  <Card 
    icon={<Languages />} 
    href="./quick-translate" 
    title="快捷翻译">
    支持快捷翻译，在软件主界面或者按下快捷键后，用户可以直接输入(或获取鼠标选中的文本)，EasyChat 将自动识别并翻译为目标语言，并支持搭配、罕见词、重点词、用法建议、词典详情等功能。
  </Card>

  <Card 
    icon={<SpellCheck2 />} 
    href="./quick-correction" 
    title="快捷纠正">
    支持快捷纠正，在软件主界面或者按下快捷键后，用户可以直接输入(或获取鼠标选中的文本)，EasyChat 将自动识别并纠正文本中的语法错误，划线标注错误的单词，并显示修改结果，同时提供修改建议和解释，帮助用户更好地理解和学习语言。
  </Card>

  <Card 
    icon={<MicSignal />} 
    href="./simultaneous-interpretation" 
    title="同声传译">
    支持同声传译功能，用户可通过实时语音识别功能显示翻译字幕，并通过麦克风说话，EasyChat 将自动识别说话内容并翻译为目标语言通过虚拟音频设备输出，达到同声传译的效果，支持 10 种语言的识别。
  </Card>
</Cards>
